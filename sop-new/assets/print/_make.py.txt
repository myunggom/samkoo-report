# -*- coding: utf-8 -*-
"""기존 사이트(sop/)의 출력 양식 4쪽을 새 사이트(sop-new/)에 출력 전용으로 옮긴다.

새 사이트 화면은 그대로 두고, '인쇄 · PDF' 를 누르면 숨은 틀 안에 이 쪽을 열어
기존 사이트와 같은 모양으로 인쇄한다. 자료는 새 사이트의 assets/data.js 를 쓴다
(같은 모양의 자료라 그대로 읽힌다). 화면에서 적은 값은 sessionStorage 'sop-print'
로 넘겨받는다.
"""
import io, os, re, shutil

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
OLD = os.path.join(ROOT, "sop")
NEW = os.path.join(ROOT, "sop-new")
PA = os.path.join(NEW, "assets", "print")
os.makedirs(PA, exist_ok=True)

for f in ["style.css", "slide.css", "inline.css", "sign.css", "day.css", "app.js", "sign.js"]:
    shutil.copy2(os.path.join(OLD, "assets", f), os.path.join(PA, f))
shutil.copy2(os.path.join(OLD, "assets", "img", "logo.png"), os.path.join(PA, "logo.png"))

# 인쇄 규칙(@media print 안쪽)만 뽑아 화면에도 적용한다.
# 숨은 틀 안에서 '인쇄될 모양 그대로' 높이를 재야 한 쪽에 맞게 줄일 수 있다.
# (기존 사이트 export.js 의 extractPrint 와 같은 방법)
def 인쇄규칙(text):
    out, i = [], 0
    while True:
        m = re.compile(r"@media\s+print\s*\{").search(text, i)
        if not m:
            break
        j, depth = m.end(), 1
        while j < len(text) and depth:
            depth += {"{": 1, "}": -1}.get(text[j], 0)
            j += 1
        out.append(text[m.end():j - 1])
        i = j
    return re.sub(r"@page[^{]*\{[^}]*\}", "", "\n".join(out))
pm = "\n".join(인쇄규칙(io.open(os.path.join(PA, f), encoding="utf-8").read())
               for f in ["style.css", "slide.css", "inline.css", "sign.css", "day.css"])
io.open(os.path.join(PA, "print-now.css"), "w", encoding="utf-8").write(
    "/* 자동 생성 — 위 css 들의 인쇄 규칙. 출력 틀에서는 화면에도 인쇄 모양을 적용한다 */\n" + pm)

# 로고 경로만 출력용 폴더로
p = os.path.join(PA, "app.js")
s = io.open(p, encoding="utf-8").read()
s = s.replace("assets/img/logo.png", "assets/print/logo.png")
io.open(p, "w", encoding="utf-8").write(s)

STUB = """<script>
/* 출력 전용 : 편집 · 이동 기능은 끈다 */
window.initInline = function () {};
window.snav = function () { return ''; };
/* 새 사이트 화면에서 적은 값 */
window.PV = function (k) {
  try {
    var d = JSON.parse(sessionStorage.getItem('sop-print') || 'null');
    if (d && d.id === qs('id') && d[k]) return d[k];
  } catch (e) {}
  return null;
};
</script>
<link rel="stylesheet" href="assets/print/print-now.css">
<style>
.btnbar, .snav, .thint { display: none !important }
</style>
<script>
/* 쪽마다 A4 가로 한 장에 맞춘다 (2026-11-19 사장 : "1페이지에 다 나오게").
   넘치는 쪽은 비율을 줄이고, 줄인 만큼 폭을 넓혀 종이 폭을 그대로 채운다. */
window.addEventListener('load', function () {
  var imgs = [].slice.call(document.images), t0 = Date.now();
  imgs.forEach(function (im) { im.loading = 'eager'; });
  (function wait() {
    if (imgs.some(function (im) { return !im.complete; }) && Date.now() - t0 < 10000)
      return setTimeout(wait, 120);
    var H = 210 / 25.4 * 96 - 4;            /* A4 가로 높이(px) - 여유 */
    [].forEach.call(document.querySelectorAll('.slide'), function (sl) {
      if (sl.offsetParent === null) return;
      sl.style.minHeight = '0';
      for (var k = 0; k < 4; k++) {
        var z = parseFloat(sl.style.zoom || 1), h = sl.offsetHeight;
        if (h <= H) break;
        z = z * H / h * 0.995;
        sl.style.zoom = z;
        sl.style.width = (297 / z) + 'mm';
      }
      sl.style.minHeight = '';
    });
    window.맞춤끝 = true;
  })();
});
</script>
"""

KEEP_CSS = {"style.css", "slide.css", "inline.css", "sign.css", "day.css"}
KEEP_JS = {"data.js": "assets/data.js", "app.js": "assets/print/app.js",
           "sign.js": "assets/print/sign.js"}

for k in ["parts", "guide", "risk", "report"]:
    s = io.open(os.path.join(OLD, k + ".html"), encoding="utf-8").read()

    def css(m):
        f = m.group(1)
        return ('<link rel="stylesheet" href="assets/print/%s">' % f) if f in KEEP_CSS else ""
    s = re.sub(r'<link rel="stylesheet" href="assets/([\w.-]+)">', css, s)

    def js(m):
        f = m.group(1)
        if f in KEEP_JS:
            out = '<script src="%s"></script>' % KEEP_JS[f]
            if f == "app.js":
                out += "\n" + STUB
            return out
        return ""
    s = re.sub(r'<script src="assets/([\w.-]+)"></script>', js, s)
    s = s.replace('<link rel="icon" href="favicon.svg">', "")

    if k == "risk":
        s = s.replace("(store.get(KEY) || {})", "(PV('risk') || store.get(KEY) || {})")
        assert "PV('risk')" in s
    if k == "report":
        s = s.replace("data = store.get(KEY) || {};", "data = PV('report') || store.get(KEY) || {};")
        assert "PV('report')" in s
    s = s.replace("<head>", "<head>\n<meta name=\"robots\" content=\"noindex\">", 1)
    io.open(os.path.join(NEW, "p-%s.html" % k), "w", encoding="utf-8").write(s)
    print("sop-new/p-%s.html" % k)
